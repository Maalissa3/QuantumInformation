\section{Lecture 1}
\subsection*{Fundamental Thermodynamics \& Equations of State}
\begin{itemize}
    \item \textbf{Ideal Gas Law:} $PV = Nk_BT = nRT$
    \item \textbf{First Law (Finite):} $\Delta U = \Delta Q - \Delta W$
    \item \textbf{First Law (Infinitesimal):} $dU = \delta Q - \delta W = TdS - PdV= dU = \sum_j \epsilon_j dn_j + \sum_j n_j d\epsilon_j$
    \item \textbf{Work:} $dW = PdV$
    \item \textbf{Heat Capacity:} $\Delta Q = C \Delta T$
\end{itemize}

\subsection*{Heat Engines \& Second Law}
\begin{itemize}
    \item \textbf{Work in Cycle:} $W = Q_2 - Q_1$
    \item \textbf{Engine Efficiency:} $\eta = \frac{W}{Q_2} = 1 - \frac{Q_1}{Q_2}$
    \item \textbf{Clausius Theorem:} $\oint \frac{dQ}{T} \leq 0$; if equal then reversible
\end{itemize}

\subsection*{Macroscopic Entropy}
\begin{itemize}
    \item \textbf{Entropy Change (Reversible, from Clausius):} $\Delta S = \int^{A}_{B;rev} \frac{dQ}{T}\leq S(A)-S(B)$
    \item \textbf{Third Law:} $S = 0$ at $T = 0$; $S(A)=\int_{0}^{T_{A}} C_{R}\frac{dT}{T}$; $S(A)\rightarrow 0$ as $T_A \rightarrow 0$
\end{itemize}

\subsection*{Boltzmann Distribution}
\begin{itemize}
    \item \textbf{Internal Energy:} $U = \sum_j n_j \epsilon_j$
    \item \textbf{Total Particles:} $N = \sum_j n_j=A\sum_j \exp(\beta \epsilon_j)$
    \item \textbf{Multiplicity:} $t(\{n_j\}) = \frac{N!}{\prod_j n_j!} \prod_j g_j^{n_j} \varpropto \Omega \{n_{j}\}$
    \item \textbf{Stirling's Approximation:} $\ln n! \approx n \ln n - n$
    \item \textbf{Boltzmann Distribution:} $n_j^* = \frac{N}{Z} \exp(\beta \epsilon_j)=\exp(\alpha+\beta\epsilon_j)$
    \item \textbf{Partition Function:} $Z = \sum_j \exp(\beta \epsilon_j)$
    \item \textbf{Occupation Numbers:} $n_j = \frac{N}{Z} \exp(\beta \epsilon_j)$
    \item \textbf{Lagrange Multiplier:} $\beta = -\frac{1}{k_BT}$
\end{itemize}

\subsection*{Microscopic Entropy}
\begin{itemize}
    \item \textbf{Boltzmann Entropy:} $S = k_B \ln \Omega= -\beta T \ln(\Omega)$
    \item \textbf{Gibbs Entropy:} $S = -k_B \sum_{j=1}^m P_j \ln P_j$; $P_j = \frac{n_j}{N}$
\end{itemize}

\subsection*{Energy Decomposition}
\begin{itemize}
    \item \textbf{Heat Component:} $\sum_j \epsilon_j dn_j$ (redistribution of particles)
    \item \textbf{Work Component:} $\sum_j n_j d\epsilon_j$ (shifting energy levels)
\end{itemize}

\subsection*{Diffusion}
\begin{itemize}
    \item \textbf{Diffusion Equation:} $\frac{\partial n}{\partial t} = D \frac{\partial^2 n}{\partial x^2}$
    \item \textbf{Gibbs Entropy (Continuous):} $S = -c \int n(x,t) \ln n(x,t) dx$
    \item \textbf{Entropy Derivative:} $\frac{dS}{dt} = cD \int \frac{1}{n}\left(\frac{\partial n}{\partial x}\right)^2 dx > 0$
    \item \textbf{Gaussian Solution:} $P(x,t) = \frac{1}{\sqrt{4\pi Dt}} \exp\left(-\frac{x^2}{4Dt}\right)$
    \item \textbf{Resulting Entropy:} $S(t) \sim \log(4\pi Dt)$ (grows logarithmically with time)
\end{itemize}

\section{Lecture 2}
\subsection*{Fundamental Measures of Information}
\begin{itemize}
    \item \textbf{Information Measure:} $f(N) = K \log N$ for $N$ equally likely messages
    \item \textbf{Information Units:} Base 2 (bits), Base 10 (dits/Hartleys), Base $e$ (nats), Base 256 (bytes)
    \item \textbf{Change of Base:} Multiply by $\log_b a$ to convert from base $a$ to base $b$
\end{itemize}

\subsection*{Discrete Noiseless Channel Capacity}
\begin{itemize}
    \item \textbf{Basic Teletype Capacity:} $H = \log_2 N$ (bits per symbol)
    \item \textbf{General Channel Capacity:} $C = \lim_{T \to \infty} \frac{\log N(T)}{T}$
    \item \textbf{Varying Symbol Durations:} $X^{-t_1} + X^{-t_2} + \dots + X^{-t_n} = 1$; $C = \log X_0$
    \item \textbf{State Constraints:} $\left| \sum_s W^{-b_{ij}^{(s)}} - \delta_{ij} \right| = 0$; $C = \log W$
\end{itemize}

\subsection*{Stochastic Sources and Probabilities}
\begin{itemize}
    \item \textbf{Equilibrium Conditions:} $P_j = \sum_i P_i p_i(j)$
    \item \textbf{Joint and Transition:} $p(i, j) = p(i)p_i(j)$
    \item \textbf{Marginal Probability:} $p(i) = \sum_j p(i, j) = \sum_j p(j)p_j(i)$
\end{itemize}

\subsection*{Shannon Entropy}
\begin{itemize}
    \item \textbf{General Entropy:} $H = -K \sum_{i=1}^n p_i \log p_i$
    \item \textbf{Binary Case:} $H = -(p \log p + q \log q)$
    \item \textbf{Recursive Rule:} $H(p_1, \dots, p_N) = H(z_1, \dots, z_M) + \sum z_i H(d_{i,1}, \dots, d_{i,r_i})$
\end{itemize}

\subsection*{Joint and Conditional Entropy}
\begin{itemize}
    \item \textbf{Joint Entropy:} $H(x, y) = -\sum_{i,j} p(i, j) \log p(i, j)$
    \item \textbf{Conditional Entropy:} $H_x(y) = -\sum_{i,j} p(i, j) \log p_i(j)$
    \item \textbf{Chain Rule:} $H(x, y) = H(x) + H_x(y)$
    \item \textbf{Independence:} $H(x, y) \leq H(x) + H(y)$ (equality if independent)
    \item \textbf{Information Non-increase:} $H(y) \geq H_x(y)$
\end{itemize}
