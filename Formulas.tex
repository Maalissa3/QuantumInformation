\section{Lecture 1}
\subsection*{Fundamental Thermodynamics \& Equations of State}
\begin{itemize}
    \item \textbf{Ideal Gas Law:} $PV = Nk_BT = nRT$
    \item \textbf{First Law (Finite):} $\Delta U = \Delta Q - \Delta W$
    \item \textbf{First Law (Infinitesimal):} $dU = \delta Q - \delta W$
    \item \textbf{Work:} $dW = PdV$
    \item \textbf{Heat Capacity:} $\Delta Q = C \Delta T$
\end{itemize}

\subsection*{Heat Engines \& Second Law}
\begin{itemize}
    \item \textbf{Work in Cycle:} $W = Q_2 - Q_1$
    \item \textbf{Engine Efficiency:} $\eta = \frac{W}{Q_2} = 1 - \frac{Q_1}{Q_2}$
    \item \textbf{Clausius Theorem:} $\oint \frac{dQ}{T} \leq 0$
\end{itemize}

\subsection*{Macroscopic Entropy}
\begin{itemize}
    \item \textbf{Entropy Change (Reversible):} $\Delta S = \int_{rev} \frac{dQ}{T}$
    \item \textbf{Third Law:} $S = 0$ at $T = 0$
\end{itemize}

\subsection*{Statistical Mechanics}
\begin{itemize}
    \item \textbf{Multiplicity:} $t(\{n_j\}) = \frac{N!}{\prod_j n_j!} \prod_j g_j^{n_j}$
    \item \textbf{Stirling's Approximation:} $\ln n! \approx n \ln n - n$
\end{itemize}

\subsection*{Boltzmann Distribution}
\begin{itemize}
    \item \textbf{Partition Function:} $Z = \sum_j \exp(\beta \epsilon_j)$
    \item \textbf{Occupation Numbers:} $n_j = \frac{N}{Z} \exp(\beta \epsilon_j)$
    \item \textbf{Lagrange Multiplier:} $\beta = -\frac{1}{k_BT}$
\end{itemize}

\subsection*{Microscopic Entropy}
\begin{itemize}
    \item \textbf{Boltzmann Entropy:} $S = k_B \ln \Omega$
    \item \textbf{Gibbs Entropy:} $S = -k_B \sum_{j=1}^m P_j \ln P_j$
    \item \textbf{Fundamental Relation:} $dU = TdS - PdV$
\end{itemize}

\subsection*{Energy Decomposition}
\begin{itemize}
    \item \textbf{Energy Change:} $dU = \sum_j \epsilon_j dn_j + \sum_j n_j d\epsilon_j$
    \item \textbf{Heat Component:} $\sum_j \epsilon_j dn_j$ (redistribution of particles)
    \item \textbf{Work Component:} $\sum_j n_j d\epsilon_j$ (shifting energy levels)
\end{itemize}
